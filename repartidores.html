package com.proyectofinal.jrmg.util; public class DotUtils { /** * Escapa caracteres especiales para etiquetas de registros en Graphviz (DOT). */ public static String escapar(String texto) { if (texto == null) { return ""; } return texto .replace("\\", "\\\\") .replace("\"", "\\\"") .replace("{", "\\{") .replace("}", "\\}") .replace("[", "\\[") .replace("]", "\\]") .replace("|", "\\|") .replace("<", "\\<") .replace(">", "\\>") .replace("\n", " ") .replace("\r", ""); } }
